# These libraries and executables use the host C/C++ toolchain. Apply IPO per
# target instead of inheriting CMAKE_INTERPROCEDURAL_OPTIMIZATION into HIP/CUDA.
set(DSV4F_CPU_TARGETS ggml-base ggml ggml-cpu llama llama-common-base llama-common
    mtmd cpp-httplib server-context llama-server-impl llama-server)
get_property(DSV4F_PROJECT_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(cpu_target IN LISTS DSV4F_PROJECT_TARGETS)
    if(cpu_target STREQUAL "dsv4f-mmq-workspace-test")
        # This executable is compiled and linked by HIP Clang.
        set_property(TARGET ${cpu_target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
        continue()
    endif()
    list(APPEND DSV4F_CPU_TARGETS ${cpu_target})
endforeach()
foreach(cpu_target IN LISTS DSV4F_CPU_TARGETS)
    if(TARGET ${cpu_target})
        set_property(TARGET ${cpu_target} PROPERTY INTERPROCEDURAL_OPTIMIZATION ${DSV4F_CPU_LTO})
    endif()
endforeach()
foreach(gpu_target IN ITEMS ggml-hip ggml-cuda)
    if(TARGET ${gpu_target})
        set_property(TARGET ${gpu_target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
    endif()
endforeach()
if(DSV4F_NATIVE)
    foreach(cpu_target IN ITEMS llama ${DSV4F_PROJECT_TARGETS})
        target_compile_options(${cpu_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-march=native>")
    endforeach()
endif()
