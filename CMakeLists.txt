cmake_minimum_required(VERSION 3.24)
project(dsv4f_inf VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/llama.cpp)
target_sources(llama PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/moe.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/mxfp4.cpp")
target_include_directories(llama PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/ggml/src")
target_link_libraries(llama PRIVATE ggml-cpu)
set_target_properties(llama-server PROPERTIES OUTPUT_NAME dsv4f-server)
add_executable(dsv4f-moe-test tests/moe.cpp src/moe.cpp src/mxfp4.cpp)
target_include_directories(dsv4f-moe-test PRIVATE src vendor/llama.cpp/ggml/src)
target_link_libraries(dsv4f-moe-test PRIVATE ggml ggml-cpu)
enable_testing()
if(TARGET ggml-hip)
    add_executable(dsv4f-prefill-attn-test tests/prefill-attn.cpp)
    target_include_directories(dsv4f-prefill-attn-test PRIVATE vendor/llama.cpp/ggml/src)
    target_link_libraries(dsv4f-prefill-attn-test PRIVATE ggml ggml-hip)
    foreach(mode IN ITEMS 1 2)
        add_test(NAME gpu-prefill-attention-mode-${mode} COMMAND dsv4f-prefill-attn-test)
        set_tests_properties(gpu-prefill-attention-mode-${mode} PROPERTIES
            ENVIRONMENT "GGML_HIP_PREFILL_FA=${mode};GGML_HIP_PREFILL_FA_DEBUG=1" TIMEOUT 180)
    endforeach()
endif()
add_executable(dsv4f-sparse-attn-test tests/sparse-attn.cpp)
target_link_libraries(dsv4f-sparse-attn-test PRIVATE ggml)

set_tests_properties(gpu-sparse-attention-parity PROPERTIES
    ENVIRONMENT "GGML_HIP_SPARSE_FA=1;GGML_HIP_SPARSE_FA_DEBUG=1" TIMEOUT 180)
add_executable(dsv4f-prefill-test tests/prefill.cpp src/moe.cpp src/mxfp4.cpp)

set_tests_properties(gpu-streamed-prefill-parity PROPERTIES TIMEOUT 180)

    add_test(NAME gpu-prefill-${mode}-parity COMMAND dsv4f-prefill-test ${mode})
    set_tests_properties(gpu-prefill-${mode}-parity PROPERTIES
        ENVIRONMENT "DSV4F_PREFILL_PROFILE=1" TIMEOUT 240)
endforeach()
add_executable(dsv4f-budget-test tests/budget.cpp)
target_include_directories(dsv4f-budget-test PRIVATE vendor/llama.cpp/ggml/src)
add_test(NAME vram-cache-budget COMMAND dsv4f-budget-test)

target_include_directories(dsv4f-cache-replay PRIVATE vendor/llama.cpp/ggml/src)
add_test(NAME complete-expert-parity COMMAND dsv4f-moe-test)
add_executable(dsv4f-graph-test tests/graphs.cpp)
target_link_libraries(dsv4f-graph-test PRIVATE ggml)
add_test(NAME gpu-graph-replay-parity COMMAND dsv4f-graph-test)
add_executable(dsv4f-service-test tests/service.cpp src/moe.cpp src/mxfp4.cpp)
target_include_directories(dsv4f-service-test PRIVATE src vendor/llama.cpp/ggml/src)
target_link_libraries(dsv4f-service-test PRIVATE ggml ggml-cpu)
add_test(NAME gpu-service-poll-parity COMMAND dsv4f-service-test poll)
add_test(NAME gpu-service-callback-parity COMMAND dsv4f-service-test callback)
add_test(NAME gpu-service-poll-zero-cache COMMAND dsv4f-service-test poll zero-cache)
add_test(NAME gpu-service-callback-zero-cache COMMAND dsv4f-service-test callback zero-cache)
add_test(NAME gpu-service-poll-headroom COMMAND dsv4f-service-test poll pressure)

set_tests_properties(gpu-service-poll-headroom gpu-service-callback-headroom PROPERTIES TIMEOUT 120)

set_tests_properties(gpu-service-scored-cache-parity PROPERTIES ENVIRONMENT "DSV4F_CACHE_POLICY=scored" TIMEOUT 120)
add_executable(dsv4f-web-http-test tests/web-http.cpp)
target_link_libraries(dsv4f-web-http-test PRIVATE llama-server-impl cpp-httplib)

set_tests_properties(web-static-assets-auth PROPERTIES TIMEOUT 30)

target_include_directories(dsv4f-cpu-probe PRIVATE src vendor/llama.cpp/ggml/src)
target_link_libraries(dsv4f-cpu-probe PRIVATE ggml ggml-cpu)

target_link_libraries(dsv4f-mxfp4-rows-test PRIVATE ggml ggml-cpu)

set_tests_properties(cpu-placement-complete-parity PROPERTIES ENVIRONMENT "DSV4F_CPU_THREADS=12;DSV4F_CPU_AFFINITY=physical;DSV4F_CPU_SCHEDULE=static;DSV4F_CPU_ROW_CHUNK=17;DSV4F_CPU_MXFP4_ROWS=4" TIMEOUT 120)

set_tests_properties(cpu-avx512-service-parity PROPERTIES ENVIRONMENT "DSV4F_CPU_THREADS=12;DSV4F_CPU_MXFP4_ROWS=4;DSV4F_CPU_MXFP4_ISA=avx512" TIMEOUT 120)
add_executable(dsv4f-kv-stream-test tests/kv-stream.cpp)

    enable_language(HIP)
    set_source_files_properties(tests/mmq-workspace.cpp PROPERTIES LANGUAGE HIP)
    add_executable(dsv4f-mmq-workspace-test tests/mmq-workspace.cpp)

    if(GGML_HIP_GRAPHS)
        target_compile_definitions(dsv4f-mmq-workspace-test PRIVATE GGML_HIP_GRAPHS)
    endif()
    target_link_libraries(dsv4f-mmq-workspace-test PRIVATE ggml ggml-hip ggml-cpu)
    add_test(NAME gpu-mmq-reserved-workspace COMMAND dsv4f-mmq-workspace-test)
    set_tests_properties(gpu-mmq-reserved-workspace PROPERTIES TIMEOUT 180)
